string MotS(object v){return v==null||v is DBNull?"":Convert.ToString(v);}
long MotL(object v){return v==null||v is DBNull?0:Convert.ToInt64(v);}
double MotD(object v){return v==null||v is DBNull?0:Convert.ToDouble(v);}
bool MotB(object v){return v!=null&&!(v is DBNull)&&Convert.ToBoolean(v);}
object MotG(Dictionary<string,object> d,string k){return d!=null&&d.ContainsKey(k)?d[k]:null;}
string MotNum(double v){return Math.Round(v,2).ToString("0.00",System.Globalization.CultureInfo.InvariantCulture);}
string MotMoneda(long id){return id==2?"USD":id==1?"EUR":"MXN";}
string MotNivel(string fuente){return MotFuentes.ContainsKey(fuente??"")?MotFuentes[fuente][0]:"";}

// Texto con variables [Nombre]. Vacío = texto por omisión. Máximo 255 (accPolizaTransaccion/accPoliza).
string MotPlantilla(string texto,string omision,Dictionary<string,string> vars){
 string t=String.IsNullOrWhiteSpace(texto)?omision:texto.Trim();
 t=System.Text.RegularExpressions.Regex.Replace(t,@"\[([A-Za-z]+)\]",m=>vars.ContainsKey(m.Groups[1].Value)?(vars[m.Groups[1].Value]??""):m.Value);
 return t.Length>255?t.Substring(0,255):t;
}

bool MotCumple(string reglas,Dictionary<string,string> valores,string partida){
 if(String.IsNullOrWhiteSpace(reglas)||reglas.Trim()=="[]")return true;
 var lista=new System.Web.Script.Serialization.JavaScriptSerializer().DeserializeObject(reglas) as object[];
 if(lista==null)throw new Exception("Las condiciones de la partida '"+partida+"' no son válidas.");
 foreach(object o in lista){
  var r=o as Dictionary<string,object>;
  string campo=MotS(MotG(r,"campo")),op=MotS(MotG(r,"operador")),valor=MotS(MotG(r,"valor")).Trim();
  if(!valores.ContainsKey(campo))throw new Exception("La partida '"+partida+"' tiene una condición sobre '"+campo+"', que no aplica a su importe.");
  string actual=valores[campo];bool ok;
  if(MotCampoTexto(campo)){
   bool igual=String.Equals(actual,valor,StringComparison.OrdinalIgnoreCase);
   if(op=="=")ok=igual;else if(op=="!=")ok=!igual;else throw new Exception("Operador '"+op+"' no válido para "+campo+" en la partida '"+partida+"'.");
  }else{
   double a=Double.Parse(actual,System.Globalization.CultureInfo.InvariantCulture),b;
   if(!Double.TryParse(valor,System.Globalization.NumberStyles.Float,System.Globalization.CultureInfo.InvariantCulture,out b))throw new Exception("Valor '"+valor+"' no numérico en la partida '"+partida+"'.");
   if(op=="=")ok=Math.Abs(a-b)<0.005;else if(op=="!=")ok=Math.Abs(a-b)>=0.005;else if(op=="<")ok=a<b;else if(op=="<=")ok=a<=b;else if(op==">")ok=a>b;else if(op==">=")ok=a>=b;
   else throw new Exception("Operador '"+op+"' no válido en la partida '"+partida+"'.");
  }
  if(!ok)return false;
 }
 return true;
}

 Func<string,string> faltaSegmento=f=>{f=(f??"").Trim();

 var docs=ctx.Query(@"SELECT d.DocumentID,ISNULL(d.FolioPrefix,N'') AS FolioPrefix,ISNULL(CONVERT(nvarchar(50),d.Folio),N'') AS Folio,

 // Renglón a partir de un importe en MXN (y su M.E.). Igual que el generador original: MXN a 2 decimales, M.E. a 6.
 Action<Dictionary<string,object>,int,double,double,string,double,Dictionary<string,string>,Dictionary<string,string>,string,List<long>> agregar=
  (p,orden,mxn,me,moneda,tc,vars,cond,documento,ds)=>{
   string nombre=MotS(MotG(p,"nombre"));double pct=MotD(MotG(p,"porcentaje"))/100.0;
   mxn*=pct;me*=pct;
   if(MotB(MotG(p,"abs"))){mxn=Math.Abs(mxn);me=Math.Abs(me);}
   if(MotB(MotG(p,"suprimirCero"))&&Math.Abs(mxn)<0.005&&Math.Abs(me)<0.005)return;
   bool cargo=MotS(MotG(p,"mov")).Equals("Cargo",StringComparison.OrdinalIgnoreCase);
   if(mxn<0||(mxn==0&&me<0)){cargo=!cargo;mxn=-mxn;me=-me;}
   var l=new MotLinea{Orden=orden,Partida=nombre,Documento=documento,Moneda=moneda,TC=tc,Concentrar=MotB(MotG(p,"concentrar")),VarsBase=vars};
   l.Docs.AddRange(ds);
   l.Omitida=!MotCumple(MotS(MotG(p,"condiciones")),cond,nombre);
   string f=MotS(MotG(p,"cuenta"));l.Cuenta=faltaSegmento(f)==""?cuentaDe(f):null;l.FaltaCuenta=l.Cuenta==null;
   l.CuentaTexto=l.Cuenta==null?(aliasDe(f)==""?f:aliasDe(f)):MotS(l.Cuenta["AliasAccountNumber"]);
   l.NombreCuenta=l.Cuenta==null?(faltaSegmento(f)!=""?"Sin cuenta: "+faltaSegmento(f):"No existe en el catálogo"):MotS(l.Cuenta["AccountName"]);
   l.Plantilla=MotS(MotG(p,"concepto"));if(l.Plantilla.Trim()=="")l.Plantilla=nombre+(documento==""?"":" - [Folio]");
   l.Concepto=MotPlantilla(l.Plantilla,"",vars);l.Referencia=MotPlantilla(MotS(MotG(p,"referencia")),"",vars);
   if(vars.ContainsKey("Folio"))l.VarsDocs.Add(vars);
   double d2=Math.Round(mxn,2),m6=Math.Round(me,6);
   if(cargo){l.Debe=d2;l.DebeME=m6;}else{l.Haber=d2;l.HaberME=m6;}
   R.Lineas.Add(l);};

 // 1) Partidas del cobro, en su orden.

  var v=new Dictionary<string,string>(vc);
  v["DocumentoID"]=MotL(d["DocumentID"]).ToString();v["Folio"]=folio;v["MonedaDocumento"]=MotMoneda(md);
  v["TotalAplicado"]=total.ToString("N2");v["TCFactura"]=tf.ToString("0.000000");v["TCPago"]=tp.ToString("0.000000");
  R.VarsDocumentos.Add(v);
  var cd=new Dictionary<string,string>(condCobro);cd["MonedaDocumento"]=v["MonedaDocumento"];cd["Retencion"]=MotNum(ret);cd["Fluctuacion"]=usd?MotNum(sgn*baseD*(tp-tf)):"0.00";
  string etiqueta=folio+" · "+v["MonedaDocumento"];var uno=new List<long>{MotL(d["DocumentID"])};
  long docID=MotL(d["DocumentID"]);
  for(int i=0;i<partidas.Count;i++){var p=partidas[i];string fu=MotS(MotG(p,"fuente"));
   if(MotNivel(fu)=="impuesto"){
    // Un renglón por cada impuesto del documento; sus condiciones eligen cuál (tasa, tipo, retención).
    for(int j=0;j<imps.Count;j++){var g=imps[j];if(MotL(g["DocumentID"])!=docID)continue;
     double monto=MotD(g["Amount"]),tasa=Math.Round(MotD(g["TaxPerc"])*100,4);
     string tasaTxt=tasa.ToString("0.####",System.Globalization.CultureInfo.InvariantCulture);
     var vi=new Dictionary<string,string>(v);vi["Impuesto"]=MotS(g["TaxName"]);vi["TipoImpuesto"]=MotS(g["TipoImpuesto"]);vi["TasaImpuesto"]=tasaTxt;
     var ci=new Dictionary<string,string>(cd);ci["ClaseImpuesto"]=MotS(g["Clase"]);ci["TipoImpuesto"]=MotS(g["TipoImpuesto"]);ci["Tasa"]=tasaTxt;
     ci["EsRetencion"]=MotB(g["Retention"])?"Si":"No";ci["Fluctuacion"]=usd?MotNum(sgn*monto*(tp-tf)):"0.00";
     int antes=R.Lineas.Count;string et=etiqueta+" · "+MotS(g["TaxName"]);
     if(fu=="IMP_FLUCT")agregar(p,i,usd?sgn*monto*(tp-tf):0,0,"MXN",1,vi,ci,et,uno);
     else{double tcImp=fu=="IMP_TCC"?tp:tf;agregar(p,i,usd?monto*tcImp:monto,usd?monto:0,usd?"USD":"MXN",usd?tcImp:1,vi,ci,et,uno);}
     if(fu!="IMP_FLUCT"&&R.Lineas.Count>antes&&!R.Lineas.Last().Omitida)cubierto.Add(j);
    }
    continue;
   }
   if(MotNivel(fu)!="documento")continue;
   // Documento en USD: M.E. = importe en dólares, MXN = dólares x TC elegido. Documento en pesos: el importe tal cual.
   double x=0,tcSel=tf;bool fx=false;
   switch(fu){
    case "DOC_TOTAL_TCF":x=total;tcSel=tf;break; case "DOC_TOTAL_TCC":x=total;tcSel=tp;break;
    case "DOC_BASE_TCF":x=baseD;tcSel=tf;break;  case "DOC_BASE_TCC":x=baseD;tcSel=tp;break;
    case "DOC_IVA_TCF":x=iva;tcSel=tf;break;     case "DOC_IVA_TCC":x=iva;tcSel=tp;break;
    case "DOC_RET_TCF":x=ret;tcSel=tf;break;     case "DOC_RET_TCC":x=ret;tcSel=tp;break;
    case "DOC_FLUCT_BASE":x=baseD;fx=true;break; case "DOC_FLUCT_IVA":x=iva;fx=true;break; case "DOC_FLUCT_TOTAL":x=total;fx=true;break;
   }
   if(fx)agregar(p,i,usd?sgn*x*(tp-tf):0,0,"MXN",1,v,cd,etiqueta,uno);
   else agregar(p,i,usd?x*tcSel:x,usd?x:0,usd?"USD":"MXN",usd?tcSel:1,v,cd,etiqueta,uno);
  }
 }
 // Impuestos con importe que ninguna partida «por impuesto» tomó (p. ej. un tipo de retención sin cuenta configurada).
 for(int j=0;j<imps.Count;j++){var g=imps[j];if(cubierto.Contains(j)||Math.Abs(MotD(g["Amount"]))<0.005)continue;
  var dd=docs.FirstOrDefault(x=>MotL(x["DocumentID"])==MotL(g["DocumentID"]));
  string fol=dd==null?MotL(g["DocumentID"]).ToString():MotS(dd["FolioPrefix"])+MotS(dd["Folio"]);
  string tipo=MotS(g["TipoImpuesto"]);if(tipo=="")tipo=MotS(g["Clase"]);
  R.ImpuestosSinPartida.Add(fol+": «"+MotS(g["TaxName"])+"» ("+tipo+", "+Math.Round(MotD(g["TaxPerc"])*100,4).ToString("0.####")+"%) por "+MotD(g["Amount"]).ToString("N2")+" no quedó en ninguna partida de impuesto.");
 }
 // 3) Concentrar cuentas: une renglones de la misma partida, cuenta, moneda y TC.
 var finales=new List<MotLinea>();
 foreach(var l in R.Lineas){
  if(!l.Concentrar||l.Omitida){finales.Add(l);continue;}
  var dest=finales.FirstOrDefault(x=>x.Concentrar&&!x.Omitida&&x.Orden==l.Orden&&x.CuentaTexto==l.CuentaTexto&&x.Moneda==l.Moneda&&Math.Abs(x.TC-l.TC)<0.0000005);
  if(dest==null){finales.Add(l);continue;}
  dest.Debe+=l.Debe;dest.Haber+=l.Haber;dest.DebeME+=l.DebeME;dest.HaberME+=l.HaberME;
  foreach(long x in l.Docs)if(!dest.Docs.Contains(x))dest.Docs.Add(x);
  dest.VarsDocs.AddRange(l.VarsDocs);
 }
 foreach(var l in finales.Where(x=>x.Concentrar&&x.VarsDocs.Count>1)){
  double neto=Math.Round(l.Debe-l.Haber,2),netoME=Math.Round(l.DebeME-l.HaberME,6);
  l.Debe=neto>0?neto:0;l.Haber=neto<0?-neto:0;l.DebeME=neto>=0?Math.Abs(netoME):0;l.HaberME=neto<0?Math.Abs(netoME):0;
  // Variables del documento: valores de todos los documentos concentrados, separados por coma.
  var vj=new Dictionary<string,string>(l.VarsBase);
  foreach(string k in MotVarsDocumento)vj[k]=String.Join(", ",l.VarsDocs.Select(x=>x.ContainsKey(k)?x[k]:"").Distinct());
  l.Concepto=MotPlantilla(l.Plantilla,"",vj);l.Documento=l.VarsDocs.Count+" documentos";
 }
 finales=finales.Where(l=>!(l.Concentrar&&!l.Omitida&&Math.Abs(l.Debe)<0.005&&Math.Abs(l.Haber)<0.005&&Math.Abs(l.DebeME)<0.005&&Math.Abs(l.HaberME)<0.005&&l.VarsDocs.Count>1)).ToList();
 R.Lineas=finales;
 // 4) Ajuste: diferencia entre cargos y abonos, sólo con el cobro totalmente aplicado y hasta $1.00.
 double sd=Math.Round(R.Lineas.Where(l=>!l.Omitida).Sum(l=>l.Debe),2),sh=Math.Round(R.Lineas.Where(l=>!l.Omitida).Sum(l=>l.Haber),2);
 double dif=Math.Round(sd-sh,2);
 bool hayAjuste=false;
 // El redondeo se ajusta con la operación totalmente aplicada, o cuando su remanente sin aplicar ya quedó
 // registrado en su propia partida (importe «sin aplicar»): lo que resta entonces es sólo redondeo.
 bool remanenteRegistrado=R.Lineas.Any(l=>!l.Omitida&&l.Orden>=0&&l.Orden<partidas.Count&&MotS(MotG(partidas[l.Orden],"fuente"))=="COBRO_SIN_APLICAR");
 bool puedeAjustar=R.AplicacionCompleta||remanenteRegistrado;
 for(int i=0;i<partidas.Count;i++){var p=partidas[i];if(MotNivel(MotS(MotG(p,"fuente")))!="ajuste")continue;hayAjuste=true;
  if(puedeAjustar&&Math.Abs(dif)>=0.01&&Math.Abs(dif)<=1.00){
   var cdif=new Dictionary<string,string>(condCobro);cdif["DiferenciaAbsoluta"]=MotNum(Math.Abs(dif));
   // Un Abono con diferencia positiva abona; si es negativa pasa a Cargo (igual que antes).
   bool esCargo=MotS(MotG(p,"mov")).Equals("Cargo",StringComparison.OrdinalIgnoreCase);
   agregar(p,i,esCargo?-dif:dif,0,"MXN",1,vc,cdif,"",new List<long>());
   var l=R.Lineas.Last();if(!l.Omitida){sd+=l.Debe;sh+=l.Haber;dif=Math.Round(sd-sh,2);}
  }
 }
 if(Math.Abs(dif)>0.005){
  if(!puedeAjustar)R.Avisos.Add("Falta aplicar "+Math.Abs(saldoSinAplicar).ToString("N2")+" "+vc["MonedaCobro"]+(esPago?" del pago. La diferencia de cargos y abonos sólo se ajusta con el pago totalmente aplicado.":" del cobro. La diferencia de cargos y abonos sólo se ajusta con el cobro totalmente aplicado."));
  else if(hayAjuste&&Math.Abs(dif)>1.00)R.Avisos.Add("La diferencia de $"+Math.Abs(dif).ToString("N2")+" supera el tope de $1.00 y no se ajusta automáticamente.");
  else if(!hayAjuste)R.Avisos.Add("El asiento no tiene una partida de «Diferencia cargos y abonos».");
 }
 foreach(var l in R.Lineas.Where(x=>!x.Omitida&&x.FaltaCuenta).GroupBy(x=>x.Partida+"|"+x.CuentaTexto).Select(g=>g.First()))
  R.Errores.Add("La partida '"+l.Partida+"' usa la cuenta "+l.CuentaTexto+": "+l.NombreCuenta.ToLowerInvariant()+".");
 R.TotalDebe=Math.Round(sd,2);R.TotalHaber=Math.Round(sh,2);
 return R;
}
